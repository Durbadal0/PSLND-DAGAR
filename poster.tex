\begin{posterblock}{The Problem: Treatment Delay in Breast Cancer}

\begin{keybox}
\centering
\textbf{Does delaying breast cancer treatment ($>$90 days) cause worse survival? How does the effect vary across geographic regions?}
\end{keybox}

\vspace{4pt}

\begin{minipage}[t]{0.47\textwidth}
{\color{FSUgarnet}\textbf{Breast Cancer Facts}}
\begin{itemize}
\item Most diagnosed cancer in US women
\item 2nd leading cause of cancer death
\item \textbf{Treatment Delay (TD):} diagnosis to first treatment
\item TD $>$ 90 days $\Rightarrow$ worse 5-year survival
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.50\textwidth}
{\color{FSUgarnet}\textbf{Florida Cancer Registry (FCR)}}
\begin{itemize}
\item $N = 68{,}009$ patients in $K = 67$ counties
\item 41.3\% right-censored
\item Treatment: $Z\!=\!1$ (TD $\leq 90$ d), $Z\!=\!0$ (TD $> 90$ d)
\item Confounders: Age, HR status, Tumor Grade, Biopsy Delay
\end{itemize}
\end{minipage}
\end{posterblock}

\vspace{4pt}

\begin{posterblock}{Key Challenges}

\begin{minipage}[t]{0.47\textwidth}
{\color{FSUgarnet}\textbf{Data Challenges}}
\begin{itemize}
\item[$\triangleright$] \textbf{Spatial dependence} among neighboring counties
\item[$\triangleright$] \textbf{Right-censoring} (41.3\%)
\item[$\triangleright$] \textbf{Confounded treatment} --- TD not random
\item[$\triangleright$] \textbf{Clustered} patients within counties
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.50\textwidth}
{\color{FSUgarnet}\textbf{Modeling Challenges}}
\begin{itemize}
\item[$\triangleright$] \textbf{Non-linear} treatment--confounder--outcome relationships
\item[$\triangleright$] \textbf{Unmeasured confounding} at county level
\item[$\triangleright$] Need \textbf{county-} and \textbf{patient-specific} effects
\item[$\triangleright$] Existing methods: no spatial + nonparametric
\end{itemize}
\end{minipage}
\end{posterblock}

\vspace{4pt}

\begin{posterblock}{Our Innovation}

\begin{keybox}
\centering
\textbf{PS-LND:} First framework combining \textbf{nonparametric causal inference} (SoftBART) with \textbf{spatial modeling} (DAGAR) for clustered survival data
\end{keybox}

\vspace{4pt}

\begin{itemize}[itemsep=4pt]
\item[\color{accentgreen}\Large$\checkmark$] \textbf{Flexible:} SBART captures smooth non-linear effects without pre-specifying interactions
\item[\color{accentgreen}\Large$\checkmark$] \textbf{Proper spatial:} DAGAR guarantees valid distribution with interpretable $\rho \in (0,1)$
\item[\color{accentgreen}\Large$\checkmark$] \textbf{No feedback:} two-stage design isolates PS from outcome model
\item[\color{accentgreen}\Large$\checkmark$] \textbf{Bayesian UQ:} full posterior with credible intervals
\item[\color{accentgreen}\Large$\checkmark$] \textbf{Multi-level:} patient, county, and state-wide causal estimates
\item[\color{accentgreen}\Large$\checkmark$] \textbf{Robust:} works under misspecification \& distributional violations
\end{itemize}

\vspace{4pt}

{\color{FSUgarnet}\textbf{Key Estimand --- CERM:}}
$$\boxed{\text{CERM}_{ij}(t^*) = \mathbb{E}\bigl[\{T_{ij}(1)\wedge t^*\} - \{T_{ij}(0)\wedge t^*\} \,\big|\, \mathbf{x}_{ij},\, \text{County}=i\bigr]}$$
\begin{center}
\textit{Expected gain in restricted mean survival: short vs.\ long TD}
\end{center}

\vspace{4pt}
\textbf{Assumptions:} SUTVA, Consistency, Unconfoundedness, Positivity, Non-informative censoring

\begin{column}{0.325\textwidth}

\begin{methodbox}{Proposed Method: PS-LND Framework}

\centering
{\small\textbf{Data} $(y,\delta,z,\mathbf{x})$\; $\longrightarrow$\; \textbf{Stage 1:} PS $\hat{e}_{ij}$\; $\longrightarrow$\; \textbf{Stage 2:} Outcome\; $\longrightarrow$\; {\color{accentgreen}\textbf{Causal Effects}}}
\raggedright

\vspace{3pt}

\begin{stagebox}{Stage 1: Propensity Score (Spatial Probit-SBART)}
\vspace{-3pt}
$$P(Z_{ij}=1\mid \mathbf{x}_{ij}, V_i) = \Phi\!\bigl(\underbrace{b(\mathbf{x}_{ij})}_{\text{SBART}} + \underbrace{V_i}_{\text{DAGAR}}\bigr)$$
\vspace{-6pt}
\begin{itemize}
\item $b(\cdot)$: unknown function --- \textbf{Soft BART prior} (smooth sigmoid splits)
\item $\mathbf{V} \sim N(\mathbf{0},\, \tau_v^{-1}\mathbf{Q}(\rho_v)^{-1})$: \textbf{DAGAR prior} for spatial confounding
\item Output: $\hat{e}_{ij} = E[\Phi\{b(\mathbf{x}_{ij}) + V_i\} \mid \mathcal{D}_1]$
\end{itemize}
\end{stagebox}

\vspace{3pt}

\begin{stagebox}{Stage 2: Outcome Model (Log-Normal AFT + BCF)}
\vspace{-3pt}
$$\log T_{ij} \sim N\!\Bigl(\underbrace{\mu(\mathbf{x}_{ij}, \hat{e}_{ij})}_{\text{prognostic SBART}} + \underbrace{\tau(\mathbf{x}_{ij})}_{\text{treatment SBART}}\!\cdot Z_{ij} + \underbrace{W_i}_{\text{DAGAR}},\;\; \sigma^2\Bigr)$$
\vspace{-6pt}
\begin{itemize}
\item $\mu(\cdot)$: baseline survival (includes $\hat{e}_{ij}$ for targeted selection)
\item $\tau(\cdot)$: heterogeneous treatment effect
\item $\mathbf{W} \sim N(\mathbf{0},\, \tau_w^{-1}\mathbf{Q}(\rho_w)^{-1})$: county-level spatial effects
\item Censored $y_{ij}$: data augmentation $\to$ sample latent $\tilde{y}_{ij}$
\end{itemize}
\end{stagebox}

\vspace{4pt}

\begin{minipage}[t]{0.47\textwidth}
{\color{accentblue}\textbf{Why SBART?}}
\begin{itemize}
\item Smooth sigmoid splits
\item No pre-specified interactions
\item Proven for causal inference
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{0.50\textwidth}
{\color{accentblue}\textbf{Why DAGAR over CAR/ICAR?}}
\begin{itemize}
\item \textbf{Proper} joint distribution
\item Positive-definite precision
\item Interpretable $\rho_w \in (0,1)$
\item Scalable to many clusters
\end{itemize}
\end{minipage}

\vspace{2pt}

{\small\textbf{Aggregation:} $\widehat{\text{ACERM}}_i = \frac{1}{n_i}\sum_j\widehat{\text{CERM}}_{ij}$;\quad $\widehat{\text{RATE}} = \frac{1}{N}\sum_{i,j}\widehat{\text{CERM}}_{ij}$}

\end{methodbox}

\vspace{4pt}
