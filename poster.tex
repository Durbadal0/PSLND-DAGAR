\begin{posterblock}{1. Big Picture \& Motivation}
\textbf{Breast cancer:} Most frequently diagnosed cancer; 2nd leading cause of cancer death among US women.

\vspace{3pt}
\textbf{Treatment Delay (TD):} Interval between diagnosis and first treatment. Delays $>$90 days significantly worsen 5-year survival rates.

\vspace{3pt}
\textbf{Florida Cancer Registry (FCR):} $N\!=\!68{,}009$ patients $\times$ $K\!=\!67$ counties; 41.3\% censoring; 6 strata (2 races $\times$ 3 stages).

\vspace{3pt}
\begin{highlightbox}
\centering
\textbf{Goal:} Estimate county-specific and patient-specific \textbf{causal effects} of short TD ($\leq$90 days) on survival while accommodating spatial association.
\end{highlightbox}
\end{posterblock}

\vspace{4pt}

\begin{posterblock}{2. Key Challenges}
\begin{enumerate}[label=\textcolor{FSUgarnet}{\textbf{\arabic*.}}]
\item \textbf{Spatial dependence} among neighboring counties
\item \textbf{Non-linear} treatment--confounder--outcome relationships
\item \textbf{Right-censored} survival data (41.3\% censoring)
\item \textbf{Non-random treatment} assignment with confounding
\item \textbf{Clustered structure:} patients nested within counties
\item Existing methods ignore spatial association or require rigid parametric forms
\end{enumerate}
\end{posterblock}

\vspace{4pt}

\begin{posterblock}{3. Proposed Method: PS-LND (SBART + DAGAR)}
A \textbf{two-stage Bayesian} approach:

\vspace{4pt}
\begin{tcolorbox}[colback=verylightblue,colframe=accentblue,arc=2mm,boxrule=1pt,title={\small\color{white}\textbf{Stage 1: Propensity Score}},colbacktitle=accentblue,top=3pt,bottom=3pt]
\vspace{-4pt}
$$P(Z_{ij}\!=\!1\mid \mathbf{x}_{ij}, V_i) = \Phi\bigl(b(\mathbf{x}_{ij}) + V_i\bigr)$$
\vspace{-8pt}
\begin{itemize}
\item $b(\cdot)$: \textbf{SBART} prior; $\mathbf{V}$: \textbf{DAGAR} spatial prior
\item Output: estimated PS $\hat{e}_{ij}$
\end{itemize}
\end{tcolorbox}

\vspace{3pt}
\begin{tcolorbox}[colback=verylightblue,colframe=accentgreen,arc=2mm,boxrule=1pt,title={\small\color{white}\textbf{Stage 2: Outcome Model (Log-Normal AFT)}},colbacktitle=accentgreen!80!black,top=3pt,bottom=3pt]
\vspace{-4pt}
$$\log T_{ij} \sim N\!\bigl(\underbrace{\mu(\mathbf{x}_{ij}, \hat{e}_{ij})}_{\text{prognostic}} + \underbrace{\tau(\mathbf{x}_{ij})}_{\text{treatment}}\!Z_{ij} + \underbrace{W_i}_{\text{spatial}},\; \sigma^2\bigr)$$
\vspace{-8pt}
\begin{itemize}
\item $\mu,\tau$: \textbf{SBART} priors (BCF structure); $\mathbf{W}$: \textbf{DAGAR} prior
\end{itemize}
\end{tcolorbox}

\vspace{4pt}
\textbf{Key Estimand --- CERM:}
\vspace{-2pt}
$$\text{CERM}_{ij}(t^*) = \mathbb{E}[\{T_{ij}(1)\wedge t^*\} - \{T_{ij}(0)\wedge t^*\} \mid \mathbf{x}_{ij}, \text{Cluster}=i]$$
\vspace{-4pt}

\textbf{Advantages:} (i) No feedback issue; (ii) SBART captures smooth nonlinearities; (iii) DAGAR guarantees proper distribution with interpretable $\rho\!\in\!(0,1)$.
\end{posterblock}

\vspace{4pt}

\begin{posterblock}{8. Conclusions \& Policy Implications}
\begin{itemize}[itemsep=3pt]
\item[\color{accentgreen}\textbullet] \textbf{PS-LND} outperforms riAFT-BART, PSW-Frailty, and IPW in all simulations
\item[\color{accentgreen}\textbullet] Short TD ($\leq$90 days) provides significant survival benefit: \textbf{7.6 months} average gain over 10 years for WA-Distant patients in FL
\item[\color{accentgreen}\textbullet] \textbf{Significant spatial heterogeneity:} effects vary from near-zero to $>$15 months across counties
\item[\color{accentgreen}\textbullet] \textbf{HR$-$ patients without biopsy delay} benefit most --- priority targets for policy intervention
\item[\color{accentgreen}\textbullet] Patients currently experiencing \textbf{long TD have the most to gain} from intervention (RATU $>$ RATT)
\item[\color{accentgreen}\textbullet] Results support \textbf{localized Patient Navigation Programs} to reduce TD in high-impact counties
\end{itemize}
\end{posterblock}
